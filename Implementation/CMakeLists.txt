cmake_minimum_required(VERSION 3.18)

set(CMAKE_CXX_STANDARD 14)
project(GaussNewtonFit)
set(SOLUTION_NAME GaussNewtonFit)
set(SOURCE_ROOT ${CMAKE_CURRENT_LIST_DIR})

if(APPLE)
	set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Mac OS X build architectures" FORCE)
endif()

if(WIN32 AND "$ENV{HOME}" STREQUAL "")
	set(HOME_ROOT $ENV{USERPROFILE})
else()
	set(HOME_ROOT $ENV{HOME})
endif()
if(WIN32)
	string(REPLACE "\\" "/" HOME_ROOT "${HOME_ROOT}")
endif()
set(WORK_ROOT ${HOME_ROOT}/Work)

if(EXISTS ${WORK_ROOT}/DevEnv/Common.cmake)
	include(${WORK_ROOT}/DevEnv/Common.cmake)
	include(${WORK_ROOT}/DevEnv/natGUI.cmake)
	include(${WORK_ROOT}/DevEnv/MatrixLib.cmake)
endif()

if(APPLE)
	set(NATPLOT_LIB_DEBUG "${MY_LIB}/natPlotD.dylib")
	set(NATPLOT_LIB_RELEASE "${MY_LIB}/natPlot.dylib")
elseif(WIN32)
	set(NATPLOT_LIB_DEBUG "${MY_LIB}/natPlotD.lib")
	set(NATPLOT_LIB_RELEASE "${MY_LIB}/natPlot.lib")
else()
	set(NATPLOT_LIB_DEBUG "${MY_LIB}/natPlotD.so")
	set(NATPLOT_LIB_RELEASE "${MY_LIB}/natPlot.so")
endif()

file(GLOB DATA_SOURCES ${SOURCE_ROOT}/src/data/*.cpp)
file(GLOB MODEL_SOURCES ${SOURCE_ROOT}/src/models/*.cpp)
file(GLOB SOLVER_SOURCES ${SOURCE_ROOT}/src/solver/*.cpp)
file(GLOB EXPERIMENT_SOURCES ${SOURCE_ROOT}/src/experiment/*.cpp)
file(GLOB REPORT_SOURCES ${SOURCE_ROOT}/src/report/*.cpp)

add_executable(gauss_newton_cli src/main.cpp ${DATA_SOURCES} ${MODEL_SOURCES} ${SOLVER_SOURCES} ${EXPERIMENT_SOURCES} ${REPORT_SOURCES})
target_include_directories(gauss_newton_cli PRIVATE ${SOURCE_ROOT}/src ${SOURCE_ROOT}/include)
target_link_libraries(gauss_newton_cli debug ${MATRIX_LIB_DEBUG} optimized ${MATRIX_LIB_RELEASE})

if(EXISTS ${WORK_ROOT}/DevEnv/Common.cmake)
	include(gaussNewtonFit.cmake)
endif()
